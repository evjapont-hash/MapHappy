# Test suite. Build with -DMAPHAPPY_BUILD_TESTS=ON and run with ctest.
#   tests/unit : no OpenGL needed, runs on every platform
#   tests/gl   : needs an OpenGL context   (-DMAPHAPPY_GL_TESTS=ON, run under a virtual display such as Xvfb)
#   tests/ui   : scripted mouse/keyboard tests of the real editor (same requirements as tests/gl)

function(maphappy_test name source)
  add_executable(${name} ${source})
  target_link_libraries(${name} PRIVATE maphappy_core)
  target_include_directories(${name} PRIVATE ${CMAKE_CURRENT_SOURCE_DIR})
  add_test(NAME ${name} COMMAND ${name})
  set_tests_properties(${name} PROPERTIES SKIP_RETURN_CODE 77 TIMEOUT 300)
  if(MAPHAPPY_TEST_VIDEO)
    set_property(TEST ${name} APPEND PROPERTY ENVIRONMENT "MAPHAPPY_TEST_VIDEO=${MAPHAPPY_TEST_VIDEO}")
  endif()
endfunction()

maphappy_test(timeline_test      unit/timeline_test.cpp)
maphappy_test(transform_test     unit/transform_test.cpp)
maphappy_test(filebrowser_test   unit/filebrowser_test.cpp)
maphappy_test(mixer_test         unit/mixer_test.cpp)
